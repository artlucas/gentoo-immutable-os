===== /etc/sudoers.d/30-managed-admins (0440 root:root) =====
# Generated by immos-managed from the policy bundle's admin.group.
# NOT by adding managed users to wheel: wheel is the LOCAL administrator's group and
# the local administrator is the way back in when the control plane is unreachable,
# unpaid for or gone (plan/19 §6.2). Keeping the two sets separate means a
# control-plane mistake cannot remove the account that would fix it.
%immos-admins ALL=(ALL:ALL) ALL
===== /etc/polkit-1/rules.d/51-managed-admins.rules (0644 root:root) =====
// Generated by immos-managed from the policy bundle's admin.group. Sits alongside
// 49-wheel.rules, which grants the same to the local administrator; both are
// consulted, so neither replaces the other.

    return ["unix-group:immos-admins"];
});
===== /etc/polkit-1/rules.d/52-managed-flatpak.rules (0644 root:root) =====
// Generated by immos-managed from policy.flatpak.install_requires_admin.
// A standard managed user gets an authentication prompt they cannot answer; an
// administrator gets one they can. allow/deny lists are carried in the bundle and
// deliberately NOT enforced in v1 — see plan/19 §6.3.
polkit.addRule(function(action, subject) {
    if (action.id.indexOf('org.freedesktop.Flatpak.app-install') === 0 ||
        action.id.indexOf('org.freedesktop.Flatpak.runtime-install') === 0 ||
        action.id.indexOf('org.freedesktop.Flatpak.app-uninstall') === 0) {
        if (subject.isInGroup("wheel") || subject.isInGroup("immos-admins")) {
            return polkit.Result.AUTH_ADMIN_KEEP;
        }
        return polkit.Result.AUTH_ADMIN;
    }
});
